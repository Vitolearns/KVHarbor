cmake_minimum_required(VERSION 3.20)
project(KVHarborServing VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
find_path(BOOST_HEADERS boost/beast.hpp REQUIRED)
find_path(JSON_HEADERS nlohmann/json.hpp REQUIRED)
option(KVHARBOR_SERVING_LLAMA "Link the pinned llama.cpp runtime" OFF)
option(KVHARBOR_SERVING_EDGE_LLM "Link the TensorRT Edge-LLM C++ SDK" OFF)
option(KVHARBOR_SERVING_AUDIO "Link the pinned sherpa-onnx C API" OFF)
set(KVHARBOR_LLAMA_BUILD "" CACHE PATH "Directory containing bin/libllama.so")
set(KVHARBOR_LLAMA_SOURCE "" CACHE PATH "Pinned llama.cpp source")
set(KVHARBOR_SHERPA_ROOT "" CACHE PATH "Pinned sherpa-onnx distribution")

add_library(kvharbor_serving_core src/runtime.cc)
target_include_directories(kvharbor_serving_core PUBLIC include "${JSON_HEADERS}")
target_link_libraries(kvharbor_serving_core PUBLIC Threads::Threads)
target_compile_options(kvharbor_serving_core PRIVATE -Wall -Wextra -Wpedantic)
if(KVHARBOR_SERVING_EDGE_LLM)
  enable_language(CUDA)
  find_package(TensorRTEdgeLLM 0.12.0 EXACT CONFIG REQUIRED)
  target_sources(kvharbor_serving_core PRIVATE src/edge_runtime.cc)
  target_link_libraries(kvharbor_serving_core PRIVATE TensorRTEdgeLLM::Core CUDA::cudart)
  target_compile_definitions(kvharbor_serving_core PRIVATE KVHARBOR_SERVING_EDGE_LLM=1)
endif()
if(KVHARBOR_SERVING_LLAMA)
  find_library(LLAMA_LIBRARY llama PATHS "${KVHARBOR_LLAMA_BUILD}/bin" NO_DEFAULT_PATH REQUIRED)
  find_library(GGML_LIBRARY ggml PATHS "${KVHARBOR_LLAMA_BUILD}/bin" NO_DEFAULT_PATH REQUIRED)
  find_library(GGML_BASE_LIBRARY ggml-base PATHS "${KVHARBOR_LLAMA_BUILD}/bin" NO_DEFAULT_PATH REQUIRED)
  find_library(LLAMA_COMMON_LIBRARY llama-common PATHS "${KVHARBOR_LLAMA_BUILD}/bin" NO_DEFAULT_PATH REQUIRED)
  target_sources(kvharbor_serving_core PRIVATE src/llama_runtime.cc)
  target_include_directories(kvharbor_serving_core PRIVATE "${KVHARBOR_LLAMA_SOURCE}/include" "${KVHARBOR_LLAMA_SOURCE}/ggml/include" "${KVHARBOR_LLAMA_SOURCE}/common")
  target_link_libraries(kvharbor_serving_core PUBLIC "${LLAMA_LIBRARY}" "${GGML_LIBRARY}" "${GGML_BASE_LIBRARY}" "${LLAMA_COMMON_LIBRARY}")
  # libggml-cuda resolves libcuda from NVIDIA Container Toolkit at runtime.
  # The build container does not need a mounted GPU driver.
  target_link_options(kvharbor_serving_core INTERFACE "-Wl,--allow-shlib-undefined")
  target_compile_definitions(kvharbor_serving_core PRIVATE KVHARBOR_SERVING_LLAMA=1)
endif()
if(KVHARBOR_SERVING_AUDIO)
  find_library(SHERPA_LIBRARY sherpa-onnx-c-api PATHS "${KVHARBOR_SHERPA_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  target_sources(kvharbor_serving_core PRIVATE src/audio_runtime.cc)
  target_include_directories(kvharbor_serving_core PRIVATE "${KVHARBOR_SHERPA_ROOT}/include")
  target_link_libraries(kvharbor_serving_core PUBLIC "${SHERPA_LIBRARY}")
  target_compile_definitions(kvharbor_serving_core PRIVATE KVHARBOR_SERVING_AUDIO=1)
endif()

add_library(kvharbor_serving_admission STATIC src/admission.cc)
target_include_directories(kvharbor_serving_admission PUBLIC "${BOOST_HEADERS}")
target_compile_definitions(kvharbor_serving_admission PUBLIC BOOST_ERROR_CODE_HEADER_ONLY=1)
target_compile_options(kvharbor_serving_admission PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(kvharbor_serving_admission PUBLIC kvharbor_serving_core)

add_executable(kvharbor-server src/server.cc)
target_compile_options(kvharbor-server PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(kvharbor-server PRIVATE kvharbor_serving_admission)

add_executable(kvharbor-runtime-bench src/runtime_bench.cc)
target_link_libraries(kvharbor-runtime-bench PRIVATE kvharbor_serving_core)

enable_testing()
add_executable(serving-core-test tests/core_test.cc)
target_link_libraries(serving-core-test PRIVATE kvharbor_serving_core)
if(KVHARBOR_SERVING_EDGE_LLM)
  # The SDK's static archives reference CUDA registration symbols. Link the
  # generated device object directly into each executable that consumes them.
  set_target_properties(kvharbor-server kvharbor-runtime-bench serving-core-test
                        PROPERTIES CUDA_RESOLVE_DEVICE_SYMBOLS ON)
endif()
add_test(NAME serving_core COMMAND serving-core-test)
add_executable(serving-engine-schedule-test tests/engine_schedule_test.cc)
target_include_directories(serving-engine-schedule-test PRIVATE src)
target_link_libraries(serving-engine-schedule-test PRIVATE Threads::Threads)
target_compile_features(serving-engine-schedule-test PRIVATE cxx_std_20)
add_test(NAME serving_engine_schedule COMMAND serving-engine-schedule-test)
set_tests_properties(serving_engine_schedule PROPERTIES TIMEOUT 15)
